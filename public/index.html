  "includeAllMatches": false
}

POST /api/lookup/file (multipart/form-data)
- file: .xlsx/.csv
- columnName: optional
- designations: optional comma-separated list
- includeAllMatches: optional true/false
- outputFormat: "xlsx" to force file export
